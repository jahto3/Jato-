Multi-Tenant Navigation

Create separate navigation for different tenants

from app.navigation.menu import MenuManager

class TenantNavigationManager:
    def __init__(self):
        self.tenant_menus = {}
    
    def get_menu_for_tenant(self, tenant_id):
        if tenant_id not in self.tenant_menus:
            self.tenant_menus[tenant_id] = MenuManager()
        return self.tenant_menus[tenant_id]
    
    def register_tenant_menu(self, tenant_id, menu_items):
        menu = self.get_menu_for_tenant(tenant_id)
        for item in menu_items:
            menu.add_item(item)

Context-Aware Menus

Show different menus based on context or conditions

from flask import g
from app.navigation import get_menu

@app.before_request
def set_user_context():
    g.user = get_current_user()
    g.user_roles = g.user.roles if g.user else []

@app.route('/page')
def page():
    menu = get_menu()
    
    # Get menu filtered by user roles
    filtered_menu = menu.get_menu(g.user_roles)
    
    return render_template('page.html', menu=filtered_menu)

Menu Caching

Cache menu structures for performance

from functools import lru_cache
from app.navigation import get_menu

class CachedMenuManager:
    def __init__(self):
        self.menu = get_menu()
        self.cache = {}
    
    def get_menu_cached(self, roles_tuple):
        if roles_tuple not in self.cache:
            self.cache[roles_tuple] = self.menu.get_menu(list(roles_tuple))
        return self.cache[roles_tuple]
    
    def clear_cache(self):
        self.cache.clear()

Dynamic Permissions

Load permissions from database

from app.navigation import get_permission_manager
from db import get_user_permissions

def load_user_permissions(user_id):
    perm_mgr = get_permission_manager()
    permissions = get_user_permissions(user_id)
    
    for permission in permissions:
        perm_mgr.add_permission(permission['role'], permission['name'])
