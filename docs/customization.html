Custom Menu Items

Create menu items with custom data

from app.navigation.menu import MenuItem

custom_item = MenuItem(
    id='custom',
    label='Custom Page',
    url='/custom',
    icon='star',  # Bootstrap icon name
    order=5,
    visible=True,
    required_roles=['user', 'admin']
)

menu.add_item(custom_item)

Dynamic Menu Creation

Create menus programmatically from database or configuration

import json
from app.navigation.menu import MenuItem, MenuManager

# Load menu structure from JSON
menu_config = json.load(open('menu.json'))
menu = MenuManager()

def create_menu_from_config(config, parent_id=None):
    for item_config in config:
        item = MenuItem(
            id=item_config['id'],
            label=item_config['label'],
            url=item_config['url'],
            icon=item_config.get('icon'),
            parent_id=parent_id,
            required_roles=item_config.get('required_roles', [])
        )
        menu.add_item(item)
        
        if 'children' in item_config:
            create_menu_from_config(item_config['children'], item.id)

create_menu_from_config(menu_config)

Custom Breadcrumbs

Override breadcrumb generation for special cases

from app.navigation import get_navigation

navigation = get_navigation()

# Custom breadcrumb for specific path
def get_custom_breadcrumb(path):
    breadcrumb = navigation.get_breadcrumb(path)
    
    # Add custom formatting or additional items
    if path.startswith('/admin'):
        breadcrumb.insert(1, {
            'label': 'Admin Panel',
            'url': '/admin',
            'active': False
        })
    
    return breadcrumb
